#!/usr/bin/env node // // deep-research.js // // A Node.js terminal-based version of your original HTML/JS "Neil's Deep Research" script, // adapted for Anthropic’s official API, preserving the same prompts & structure. // // NOTE: The model name "claude-3-7-sonnet-20250219" may not be publicly available. // Replace it with an Anthropic model your key can access if needed. // const readline = require('readline'); const fetch = require('node:fetch'); // In Node 18+, globalThis.fetch is built in, but let's be explicit // ─────────── Terminal Colors ─────────── const RESET = "\x1b[0m"; const BRIGHT = "\x1b[1m"; const FG_GREEN = "\x1b[32m"; const FG_CYAN = "\x1b[36m"; const FG_YELLOW = "\x1b[33m"; const FG_RED = "\x1b[31m"; // ─────────── Original Prompts ─────────── const QUERY_OPTIMIZATION_PROMPT = `You are an expert academic research assistant specializing in transforming casual questions into precise academic search queries. Your task is to: 1. Analyze the user's question and identify the core research topics 2. Reformulate it using academic terminology and precision 3. Optimize it for scientific literature search 4. The query should be optimized for bioRxiv searches. Example: Input: "How do brains create memories?" Output: "Neural mechanisms of memory formation and consolidation in hippocampal and cortical circuits"`; const PAPER_SELECTION_PROMPT = `You are a research paper selection algorithm. You'll receive a scientific query and a list of paper titles. Your task is to select the 5 most relevant papers based on their titles, authors, and publication dates. You should consider: 1. Relevance to the query topic 2. Recency of publication (newer is generally better) 3. Comprehensiveness of approach 4. Methodological soundness (based on what can be inferred from the title) 5. Preference for reviews or papers that appear to synthesize multiple approaches Return ONLY a JSON array of DOIs for the 5 most relevant papers. Example: ["10.1101/123456", "10.1101/234567", "10.1101/345678", "10.1101/456789", "10.1101/567890"] Do not include any explanation or commentary, just the JSON array.`; const BEST_PAPER_SELECTION_PROMPT = `You are a scientific literature expert. You'll receive a research query and abstracts for several papers. Your task is to select the SINGLE best paper that would be most informative for addressing the research query. Consider: 1. Direct relevance to the specific research question 2. Comprehensiveness of approach 3. Methodological soundness 4. Clarity and informativeness of the abstract 5. Potential for providing definitive or novel insights Return ONLY the DOI of the best paper. No explanation or commentary.`; const RESEARCH_SYNTHESIS_PROMPT = `You are Deep Research, an advanced scientific research assistant powered by Claude 3.7. You specialize in synthesizing research papers into comprehensive, accurate reports tailored to the user's original question. You will receive: 1. The user's original query 2. An academically reformulated version of the query 3. Details about a scientific paper, including its abstract Your task is to: 1. Analyze the paper in relation to the research question 2. Synthesize its findings as if you had read the full paper 3. Create a comprehensive research report that: - Addresses the original question directly - Explains key concepts and methodologies - Summarizes the main findings and their implications - Adopts a scholarly but accessible tone - Includes relevant context from your general knowledge - Acknowledges limitations where appropriate Format your response as a well-structured research report with: - A concise introduction explaining the research question - Main sections covering key aspects of the topic - A conclusion summarizing the findings - References to the paper you "analyzed" Important: Write as if you have thoroughly read and analyzed the complete paper, not just the abstract. Use your knowledge to infer what the full paper likely contains based on the abstract and typical structure of papers in this field.`; // ────────────────────────────────────────── // Terminal Input Helpers // ────────────────────────────────────────── const rl = readline.createInterface({ input: process.stdin, output: process.stdout }); // Ask a question in the terminal. If hideInput=true, mask typed characters. function ask(question, hideInput = false) { return new Promise((resolve) => { if (!hideInput) { rl.question(question, (answer) => resolve(answer.trim())); } else { process.stdin.on("data", (char) => { char = char + ""; switch (char) { case "\n": case "\r": case "\u0004": process.stdout.write("\n"); break; default: process.stdout.write("\x1B[2K\x1B[200D" + question + Array(rl.line.length + 1).join("*")); break; } }); rl.question(question, (answer) => resolve(answer.trim())); } }); } // ────────────────────────────────────────── // Utility: Build Anthropic-Style Prompt // (Because we can’t use a messages[] array with role: 'system'/'user'.) // ────────────────────────────────────────── function buildAnthropicPrompt(systemPrompt, userContent) { // Minimal approach: treat systemPrompt as an initial "Human" instruction // Then userContent as the next "Human" message, with a filler Assistant line in between return `Human: ${systemPrompt}\n\nAssistant:\n\nHuman: ${userContent}\n\nAssistant:`; } // ────────────────────────────────────────── // Replacing the old "messages" approach with Anthropic's single-prompt approach // ────────────────────────────────────────── async function callClaudeAPI(apiKey, systemPrompt, userContent, maxTokens = 1000) { try { // Build the final "Human: ... Assistant: ... Human: ... Assistant:" prompt const prompt = buildAnthropicPrompt(systemPrompt, userContent); // Call Anthropic’s official API const response = await fetch('https://api.anthropic.com/v1/complete', { method: 'POST', headers: { 'Content-Type': 'application/json', 'x-api-key': apiKey, 'anthropic-version': '2023-06-01' // from your original script }, body: JSON.stringify({ model: 'claude-3-7-sonnet-20250219', // same as your original script prompt, max_tokens_to_sample: maxTokens, temperature: 0.7, top_k: 0, top_p: 1, stream: false, stop_sequences: ["Human:"] }) }); if (!response.ok) { let errorInfo; try { const errData = await response.json(); errorInfo = errData.error?.message || `Status: ${response.status}`; } catch (e) { errorInfo = `Status: ${response.status}`; } throw new Error(`API Error: ${errorInfo}`); } // Parse the JSON const jsonData = await response.json(); // The text completion is in jsonData.completion return jsonData; // We'll return the entire object } catch (error) { console.error(`${FG_RED}Claude API error:${RESET}`, error); throw new Error(`Failed to call Claude API: ${error.message}`); } } // ────────────────────────────────────────── // Keep your original helper functions // ────────────────────────────────────────── function extractSearchTerms(optimizedQuery) { // Clean query and split into words const words = optimizedQuery .toLowerCase() .replace(/[^\w\s-]/g, '') .split(/\s+/); // Filter out common stop words const stopWords = ['the','and','of','in','on','at','to','for','with','by','as','that','this','are','is']; const filteredWords = words.filter(word => word.length > 2 && !stopWords.includes(word)); // Get top 5 longest words const sortedWords = filteredWords.sort((a, b) => b.length - a.length).slice(0, 5); // Join with OR for bioRxiv return sortedWords.join(' OR '); } // ────────────────────────────────────────── // Data flow functions // ────────────────────────────────────────── async function processInput(userQuery, apiKey) { return { query: userQuery.trim(), apiKey: apiKey.trim(), timestamp: new Date().toISOString() }; } async function optimizeQuery(processedInput) { try { // callClaudeAPI -> returns { completion: "...", ... } const result = await callClaudeAPI( processedInput.apiKey, QUERY_OPTIMIZATION_PROMPT, processedInput.query, 300 ); // Instead of result.content[0].text, we use result.completion const optimizedQuery = result.completion; return { originalQuery: processedInput.query, optimizedQuery, apiKey: processedInput.apiKey, searchTerms: extractSearchTerms(optimizedQuery) }; } catch (error) { throw new Error(`Failed to optimize query: ${error.message}`); } } async function retrievePapers(queryData) { try { const currentYear = new Date().getFullYear(); const searchUrl = `https://api.biorxiv.org/details/biorxiv/2021-01-01/${currentYear}-12-31/${encodeURIComponent(queryData.searchTerms)}/0/50`; console.log(`${FG_CYAN}Searching bioRxiv with URL:${RESET} ${searchUrl}`); const searchResponse = await fetch(searchUrl); if (!searchResponse.ok) { throw new Error(`BioRxiv API error: ${searchResponse.status}`); } const searchData = await searchResponse.json(); if (!searchData.collection || searchData.collection.length === 0) { return { success: false, error: "No papers found. Try different search terms." }; } // Step 2: Format paper titles const paperTitles = searchData.collection.map(paper => ({ doi: paper.doi, title: paper.title, authors: paper.authors, date: paper.date })); // Step 3: Ask Claude to select promising papers const titleSelectionResult = await callClaudeAPI( queryData.apiKey, PAPER_SELECTION_PROMPT, JSON.stringify({ query: queryData.optimizedQuery, papers: paperTitles }), 300 ); // The actual text from Anthropic is in titleSelectionResult.completion let textContent = titleSelectionResult.completion; let selectedDOIs; try { const jsonMatch = textContent.match(/\[.*\]/s); if (jsonMatch) { selectedDOIs = JSON.parse(jsonMatch[0]); } else { throw new Error("Could not parse DOI list from Claude's response"); } } catch (error) { console.error("Error parsing DOIs:", error); selectedDOIs = paperTitles.slice(0, 3).map(p => p.doi); } if (!Array.isArray(selectedDOIs) || selectedDOIs.length === 0) { throw new Error("Failed to get valid paper selections"); } // Step 5: Fetch abstracts const abstractPromises = selectedDOIs.map(doi => fetch(`https://api.biorxiv.org/details/biorxiv/${doi}`) .then(res => res.ok ? res.json() : null) .then(data => { if (data && data.collection && data.collection.length > 0) { return { doi: data.collection[0].doi, title: data.collection[0].title, authors: data.collection[0].authors, abstract: data.collection[0].abstract, url: `https://www.biorxiv.org/content/${data.collection[0].doi}v${data.collection[0].version}` }; } return null; }) .catch(err => { console.error(`Error fetching abstract for DOI ${doi}:`, err); return null; }) ); const abstracts = (await Promise.all(abstractPromises)).filter(a => a !== null); if (abstracts.length === 0) { return { success: false, error: "Failed to retrieve paper abstracts" }; } // If only one paper if (abstracts.length === 1) { return { success: true, query: queryData, allPaperTitles: paperTitles, selectedAbstracts: abstracts, bestPaper: abstracts[0] }; } // Step 7: select best paper const bestPaperResult = await callClaudeAPI( queryData.apiKey, BEST_PAPER_SELECTION_PROMPT, JSON.stringify({ query: queryData.optimizedQuery, papers: abstracts }), 300 ); let bestPaperDOI = bestPaperResult.completion.trim(); if (bestPaperDOI.includes("10.1101/")) { const doiMatch = bestPaperDOI.match(/10\.1101\/[0-9v.]+/); if (doiMatch) bestPaperDOI = doiMatch[0]; } const bestPaper = abstracts.find(p => p.doi === bestPaperDOI); if (!bestPaper) { console.warn("Best paper not found in abstracts, using first paper instead"); return { success: true, query: queryData, allPaperTitles: paperTitles, selectedAbstracts: abstracts, bestPaper: abstracts[0] }; } return { success: true, query: queryData, allPaperTitles: paperTitles, selectedAbstracts: abstracts, bestPaper }; } catch (error) { console.error(`${FG_RED}Paper retrieval error:${RESET}`, error); return { success: false, error: `Failed to retrieve papers: ${error.message}` }; } } async function synthesizeResearch(paperData) { try { // Create context const synthesisContext = { originalQuery: paperData.query.originalQuery, optimizedQuery: paperData.query.optimizedQuery, paper: paperData.bestPaper }; const synthesisResult = await callClaudeAPI( paperData.query.apiKey, RESEARCH_SYNTHESIS_PROMPT, JSON.stringify(synthesisContext), 4000 ); // The final text is in .completion return { originalQuery: paperData.query.originalQuery, optimizedQuery: paperData.query.optimizedQuery, paper: paperData.bestPaper, synthesis: synthesisResult.completion, timestamp: new Date().toISOString() }; } catch (error) { console.error(`${FG_RED}Research synthesis error:${RESET}`, error); throw new Error(`Failed to synthesize research: ${error.message}`); } } // ────────────────────────────────────────── // (Optional) Minimal "render" in console // ────────────────────────────────────────── function renderResults(researchData) { console.log(`${BRIGHT}${FG_GREEN}\n┌────────────────────────────────────────┐`); console.log(`│ DEEP RESEARCH REPORT │`); console.log(`└────────────────────────────────────────┘${RESET}`); console.log(`${FG_CYAN}\nQUERY:${RESET} ${researchData.originalQuery}`); console.log(`${FG_CYAN}\nACADEMIC FORMULATION:${RESET} ${researchData.optimizedQuery}`); console.log(`${FG_CYAN}\nPRIMARY SOURCE:${RESET}`); console.log(`Title: ${researchData.paper.title}`); console.log(`Authors:${researchData.paper.authors}`); console.log(`DOI: ${researchData.paper.doi}`); console.log(`URL: ${researchData.paper.url}`); console.log(`${FG_CYAN}\nRESEARCH SYNTHESIS:${RESET}`); console.log(researchData.synthesis); console.log(`${FG_GREEN}\nReport generated at ${new Date().toLocaleString()}${RESET}\n`); } // ────────────────────────────────────────── // Main driver function (replaces DOM button click) // ────────────────────────────────────────── async function main() { console.log(`${BRIGHT}${FG_GREEN} ╔══════════════════════════════════════╗ ║ Neil's Deep Research ║ ║ Powered by Claude 3.7 ║ ╚══════════════════════════════════════╝${RESET}\n`); try { const apiKey = await ask(`${FG_YELLOW}Claude API Key: ${RESET}`, true); const query = await ask(`${FG_YELLOW}Research Question: ${RESET}`); if (!apiKey || !query) { console.error(`${FG_RED}Please provide both an API key and a research question.${RESET}`); process.exit(1); } // Step 1: Process input const processedInput = await processInput(query, apiKey); // Step 2: Optimize query console.log(`${FG_CYAN}\nOptimizing query...${RESET}`); const optimizedQuery = await optimizeQuery(processedInput); // Step 3: Retrieve papers console.log(`${FG_CYAN}Retrieving relevant papers...${RESET}`); const paperData = await retrievePapers(optimizedQuery); if (!paperData.success) { throw new Error(paperData.error || 'Failed to retrieve papers'); } // Step 4: Synthesize research console.log(`${FG_CYAN}Synthesizing research report...${RESET}`); const researchResults = await synthesizeResearch(paperData); // Step 5: Render renderResults(researchResults); } catch (error) { console.error(`${FG_RED}Research Error:${RESET} ${error.message}`); } finally { rl.close(); } } // Run the main function main();
